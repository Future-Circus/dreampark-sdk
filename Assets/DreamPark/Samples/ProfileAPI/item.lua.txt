-- award-item.lua
-- Add or remove an item in the bound player's profile inventory.
--
-- Designed to drop onto either a LuaBehaviour OR an EasyLua node with no
-- code changes:
--   • LuaBehaviour — does NOT auto-fire. Call from another Lua script
--     via a ScriptInjection:
--         awardItem.trigger()          -- use the @var-configured amount
--         awardItem.trigger(-1)        -- override amount per-call
--   • EasyLua — fires automatically when the EasyEvent chain reaches
--     this node, then passes control to the next event the same frame.
--
-- `amount` controls direction:
--     amount > 0   → award N copies (or 1 unique entry if makeUnique=true)
--     amount < 0   → remove |N| copies
--     amount == 0  → no-op
--
-- Unique items with per-instance METADATA (Oct 2026) — a caught creature,
-- a rolled sword — are shown further down: awardWithMetadata(meta),
-- levelUp(instanceId), list(). Each instance is its own inventory row with
-- its own table of your data, which you can change later with updateItem.
--
-- `makeUnique` produces a metadata-tagged inventory entry. Every call
-- creates a distinct row keyed by `<itemId>_<random>`, so the same
-- catalog id can stack as multiple unique instances. The backend forces
-- amount=1 for unique entries. Has no effect when amount is negative.
--
-- NOTE: makeUnique is for loot where each copy is meant to be
-- DISTINGUISHABLE (procedurally rolled gear, numbered collectibles). It is
-- NOT how you make an item non-stacking — every call mints another row, and
-- each row is worth profile XP. For a one-of-a-kind item (a wand, a key, a
-- costume), untick "Stacks" on the catalog entry in the developer portal
-- (or on the item's card in Content Uploader → Items) instead; awardItem
-- then becomes a no-op once the player has it. A unique/metadata item must
-- KEEP Stacks ticked: a non-stacking item is a singleton and the server
-- ignores award metadata for it.
--
-- The item's catalog entry (title, icon, type, rarity, Stacks) is authored
-- in Content Uploader → Park Assets → Items: every id this script passes is
-- found there automatically (the @var's Inspector value included) and is
-- pushed to the backend with Push Items to Portal or with every upload.
--
--
-- Requires the player to have ENTERED one of your attractions. Awards made
-- before they walk into a GameArea of this game are held and sent the moment
-- they do — so wiring this to a park-wide script that fires on start() will
-- look like nothing happened until the player actually arrives.
-- @var itemId string ""
-- @var amount int 1
-- @var makeUnique bool false

local UE = CS.UnityEngine
local Debug = UE.Debug

-- ── State ────────────────────────────────────────────────────────
local fired = false

-- ── Runtime detection ────────────────────────────────────────────
-- EasyLua exposes a public `delayNextEvent` bool field; LuaBehaviour
-- does not. We probe via pcall AND require a boolean result, so the
-- check stays correct whether XLua throws on the missing member
-- (older versions) or silently returns nil (newer ones).
local function is_easy_lua()
    if self == nil then return false end
    local ok, val = pcall(function() return self.delayNextEvent end)
    return ok and type(val) == "boolean"
end

-- ── Public API ───────────────────────────────────────────────────
function trigger(overrideAmount)
    local id = itemId
    if id == nil or id == "" then
        Debug.LogWarning("[award-item] itemId not set — nothing to do")
        fired = true
        return false
    end
    local amt = overrideAmount or amount or 1
    if amt == 0 then
        fired = true
        return false
    end

    if amt > 0 then
        pcall(function()
            if makeUnique then
                dp.profile.awardUniqueItem(id)
            else
                dp.profile.awardItem(id, amt)
            end
        end)
        Debug.Log("[award-item] +" .. tostring(amt) .. " '" .. tostring(id) .. "'"
            .. (makeUnique and " (unique)" or ""))
    else
        local toRemove = -amt
        pcall(function()
            dp.profile.removeItem(id, toRemove)
        end)
        Debug.Log("[award-item] -" .. tostring(toRemove) .. " '" .. tostring(id) .. "'")
    end

    fired = true
    return true
end

-- ── Unique items with metadata ───────────────────────────────────
-- Metadata is your own table: strings, numbers, booleans and nested tables,
-- at most 4 KB of JSON per item. The SDK adds `contentId` (this game) for you.
-- Keep ids as strings. Writes share a budget of 60 per minute per player, so
-- update on game events (a battle won), never every frame.

-- awardWithMetadata({ level = 1, nickname = 'Rusty' }) → mints ONE new
-- instance. The callback gets the exact row written; keep its instanceId,
-- that's the key for updateItem / removeItem later.
function awardWithMetadata(meta)
    local id = itemId
    if id == nil or id == "" then return false end
    if not dp.profile.supportsItemMetadata then
        dp.profile.awardUniqueItem(id)              -- older SDK: no metadata
        return true
    end
    dp.profile.awardUniqueItem(id, meta or { level = 1, xp = 0 }, function(ok, item)
        if not ok or item == nil then
            Debug.LogWarning("[award-item] unique award failed for '" .. tostring(id) .. "'")
            return
        end
        Debug.Log("[award-item] minted " .. tostring(item.instanceId)
            .. " level " .. tostring(item.metadata and item.metadata.level))
    end)
    return true
end

-- levelUp(instanceId) → merges { level = level + 1 } into that instance's
-- metadata. Other keys are kept; pass replace = true as the 4th argument of
-- updateItem to swap the whole table instead.
function levelUp(instanceId)
    local row = dp.profile.getItem(instanceId)
    local level = (row and row.metadata and row.metadata.level) or 1
    dp.profile.updateItem(instanceId, { level = level + 1 }, function(ok, item)
        Debug.Log("[award-item] levelUp " .. tostring(instanceId) .. " ok=" .. tostring(ok)
            .. " level=" .. tostring(item and item.metadata and item.metadata.level))
    end)
end

-- list() → every instance of this item the player owns, each with its own
-- metadata. getItems also takes { baseItemId =, type =, contentId = }.
function list()
    local rows = dp.profile.getItems(itemId)
    for i, it in ipairs(rows) do
        Debug.Log("[award-item] #" .. i .. " " .. tostring(it.instanceId)
            .. " level=" .. tostring(it.metadata and it.metadata.level))
    end
    return rows
end

-- ── Lifecycle ────────────────────────────────────────────────────
function start()
    if is_easy_lua() then
        trigger()
    end
end

function update()
    if fired and is_easy_lua() then
        return true
    end
end
